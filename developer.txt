#### create a virtual python environment 
python3 -m venv .venv
or
python3 -m venv 'directory'

#### activate virtual python environment
#### use activate.ps1 when in powershell
.venv\Scripts\activate.bat
or
.directory\Scripts\activate.bat

#### to exit virtual environment
deactivate

#### run pytest scripts
poetry run pytest -v (shows name of each test)
poetry run pytest -vs (shows name and print statements)
poetry run pytest -v -k 'test_XX' (run a specific function)

#### create new poetry project
poetry new 'projectname'

#### to update dependencies / update the dependencies as according to the pyproject.toml file.
poetry update

#### to install the project itself
poetry install

#### add new dependency to the project
poetry add 'dependency'
#### to remove use
poetry remove 'dependency'

#### list existing python libraries
pip list
